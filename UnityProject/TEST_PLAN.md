## 自動テストの実装（今後の予定）

### Unity Test Framework

```csharp
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;
using System.Collections;

public class PlayerTests
{
    [Test]
    public void Player_InitialStats_AreCorrect()
    {
        // Arrange
        GameObject playerObj = new GameObject();
        Player player = playerObj.AddComponent<Player>();

        // Act
        player.Initialize();

        // Assert
        Assert.AreEqual(100, player.MaxHP);
        Assert.AreEqual(10, player.Attack);
        Assert.AreEqual(5, player.Defense);
    }

    [UnityTest]
    public IEnumerator Player_TakeDamage_ReducesHP()
    {
        // Arrange
        GameObject playerObj = new GameObject();
        Player player = playerObj.AddComponent<Player>();
        player.Initialize();
        int initialHP = player.CurrentHP;

        // Act
        player.TakeDamage(20);
        yield return null;

        // Assert
        Assert.AreEqual(initialHP - 20, player.CurrentHP);
    }
}
```
